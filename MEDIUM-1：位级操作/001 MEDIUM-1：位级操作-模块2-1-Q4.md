```c
#include <stdio.h>
int main()
{
    int number;
    int i=1;
    printf("请输入一个整数：");
    scanf("%d",&number);
    int temp=number;
    if(temp==0)
    {
        printf("找不到");
    }
    else
    {
        while((temp&1)!=1)
        {
            temp>>=1;
            i++;
        }
        printf("%d的数%d",number,i);
    }
    return 0;
}
```
