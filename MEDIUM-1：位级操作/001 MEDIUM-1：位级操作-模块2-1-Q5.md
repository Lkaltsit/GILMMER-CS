#include <stdio.h>
#include <string.h>
int main()
{
    char a[105],b[105],temp[105];
    printf("请输入第一个正整数：");
    scanf("%s",a);
    printf("请输入第二个正整数：");
    scanf("%s",b);
    if(strcmp(a,b)>=0)
    {
        strcpy(temp,a);
        strcpy(a,b);
        strcpy(b,temp);
    }
    memset(temp,0,sizeof(temp));
    for(int j=0;j<strlen(a);j++)
    {
        temp[105-strlen(a)+j]=a[j];
    }
    strcpy(a,temp+(105-strlen(a)));
    int i=0;
    while(1)
    {
        if(b[i]=='\0')
        {
            break;
        }
        else
        {
            if(a[i]==b[i]&&a[i]=='1')
            {
                printf("1");
            }
            else if(a[i]==b[i]&&a[i]=='0')
            {
                printf("0");
            }
            else
            {
                for(int k=0;k<strlen(b)-i;k++)
                {
                    printf("0");
                }
                break;
            }
        }
        i++;
    }
    return 0;

}
